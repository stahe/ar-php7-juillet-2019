    13.1. إنشاء قاعدة البيانات [dbimpots-2019]

    باتباع المثال الوارد في الفقرة المرتبطة، نقوم بإنشاء قاعدة بيانات MySQL باسم [dbimpots-2019]، وسيكون مالكها هو [admimpots] بكلمة مرور [mdpimpots]:

    • في [1-4] أعلاه، نرى قاعدة البيانات [dbimpots-2019] التي لا تحتوي على أي جداول في الوقت الحالي؛

    • في [1-5] أعلاه، نرى أن المستخدم [admimpots] لديه جميع الحقوق على قاعدة البيانات [dbimpots-2019]. ما لا نراه هنا هو أن هذا المستخدم لديه كلمة المرور [admimpots]؛

    نقوم الآن بإنشاء الجدول [tbtranches] الذي سيحتوي على شرائح الضرائب:

    • في [1-7]، نقوم بإنشاء جدول باسم [tbtranches] يحتوي على 4 أعمدة؛

    • في [3-6]، نحدد عمودًا باسم [id] (3)، من النوع الصحيح [int] (4)، والذي سيكون المفتاح الأساسي [6] للجدول وسيتم زيادة قيمته تلقائيًا [5] بواسطة SGBD. وهذا يعني أن MySQL ستتولى بنفسها إدارة قيم المفتاح الأساسي عند عمليات الإدراج. وستعيّن القيمة 1 للمفتاح الأساسي في عملية الإدراج الأولى، ثم 2 في العملية التالية، وهكذا دواليك؛
    • في [7]، يعرض علينا المساعد خيارات أخرى لتكوين المفتاح الأساسي. هنا نكتفي بتأكيد [7] القيم الافتراضية؛

    • في [8-16]، نحدد الأعمدة الثلاثة الأخرى في الجدول:
      • [limites] (8) من النوع العشري (9) المكون من 10 أرقام منها رقمان عشريان (10) سيحتوي على عناصر العمود 17 الخاص بشرائح الضرائب؛
      • [coeffR] (11) من النوع العشري (12) مكون من 6 أرقام منها رقمان عشريان (13) سيحتوي على عناصر العمود 18 من شرائح الضرائب؛
      • [coeffN] (14) من نوع الرقم العشري (15) المكون من 10 أرقام منها رقمان عشريان (16) سيحتوي على عناصر العمود 19 من شرائح الضرائب؛

    بعد التحقق من صحة هذه البنية، نحصل على النتيجة التالية:

    • في [5]، تشير أيقونة المفتاح إلى أن العمود [id] هو مفتاح أساسي. ونلاحظ أيضًا أن هذا المفتاح الأساسي له قيم صحيحة (6) وأنه يُدار (يتم زيادة قيمته تلقائيًا) بواسطة MySQL؛

    وبنفس الطريقة التي أنشأنا بها الجدول [tbtranches]، نقوم بإنشاء الجدول [tbconstantes] الذي سيحتوي على الثوابت الخاصة بحساب الضريبة:

    13.2. تنظيم الكود

    لتوضيح دور البرامج النصية المختلفة PHP التي نكتبها بشكل أفضل، سنقوم بتنظيم الكود في مجلدات:

    • في [1]، نظرة عامة على الإصدار 05؛
    • في [2]، كيانات التطبيق، الكيانات المتبادلة بين الطبقات؛
    • في [3]، أدوات التطبيق؛
    • في [4]، البيانات المستخدمة أو التي ينتجها التطبيق. وقد قررنا هنا استخدام ملفات jSON فقط للملفات النصية. وتتميز هذه الملفات بعدة مزايا:
      • تتعرف عليها العديد من الأدوات؛
      • تتميز هذه الأدوات بخاصية تمييز الألوان النحوية. علاوة على ذلك، فإن صيغة jSON تخضع لقواعد معينة. وعندما لا يتم الالتزام بهذه القواعد، تقوم الأدوات بالإبلاغ عن ذلك. على سبيل المثال، من الأخطاء التي يصعب اكتشافها في ملف نصي بسيط استخدام الحرف O كبير أو صغير بدلاً من الأصفار. إذا حدث هذا الخطأ، فسيتم الإبلاغ عنه. فعلى سبيل المثال، في الكود jSON:

    "plafondRevenusCouplePourReduction": 42O74

    حيث تم عن غير قصد وضع حرف O كبير بدلاً من الصفر في [42074]، يقوم NetBeans بالإبلاغ عن الخطأ:

    في الواقع، يتعرف NetBeans على الحرف O الكبير الذي يجعل من [49O74] سلسلة أحرف. ويستنتج من ذلك أن الصيغة الصحيحة يجب أن تكون [4-5]: أي أن السلسلة [47O74] يجب أن تكون محاطة بعلامات اقتباس. وبالتالي، يتم لفت انتباه المطور إلى الخطأ ويمكنه تصحيحه: إما بإضافة علامتي الاقتباس، أو باستبدال الحرف «O» بالرقم صفر؛

    العناصر الأخرى في الإصدار 05 هي كما يلي:

    • في [6]، واجهات وفئات الطبقة [Dao]؛
    • في [7]، واجهات وفئات الطبقة [métier]؛
    • في [8]، البرامج النصية الرئيسية للإصدار 05؛

    13.3.2.1. الفئة الأساسية [BaseEntity]

    لتبسيط كود الكيانات، سنتبع القاعدة التالية: تحمل سمات الكيان نفس أسماء سمات الملف jSON الذي يجب أن يقوم الكيان بتغليفه. بموجب هذه القاعدة، فإن الكيانات [Database, TaxAdminData, TaxPayerData] لها نقاط مشتركة يمكن تحويلها إلى فئة أصلية. وستكون هذه هي الفئة [BaseEntity] التالية:

    
<?php

namespace Application;

class BaseEntity {
  // السمة
  protected $arrayOfAttributes;

  // التهيئة من ملف jSON
  public function setFromJsonFile(string $jsonFilename) {
    // يتم استرداد محتوى ملف البيانات الضريبية
    $fileContents = \file_get_contents($jsonFilename);
    $erreur = FALSE;
    // خطأ؟
    if (!$fileContents) {
      // تسجيل الخطأ
      $erreur = TRUE;
      $message = "Le fichier des données [$jsonFilename] n'existe pas";
    }
    if (!$erreur) {
      // استرداد الرمز jSON من ملف التكوين في جدول مرتبط
      $this->arrayOfAttributes = \json_decode($fileContents, true);
      // خطأ؟
      if ($this->arrayOfAttributes === FALSE) {
        // يتم تسجيل الخطأ
        $erreur = TRUE;
        $message = "Le fichier de données jSON [$jsonFilename] n'a pu être exploité correctement";
      }
    }
    // خطأ؟
    if ($erreur) {
      // يتم إثارة استثناء
      throw new ExceptionImpots($message);
    }
    // تهيئة سمات الفئة
    foreach ($this->arrayOfAttributes as $key => $value) {
      $this->$key = $value;
    }
    // إرجاع الكائن
    return $this;
  }

  public function checkForAllAttributes() {
    // التحقق من تهيئة جميع المفاتيح
    foreach (\array_keys($this->arrayOfAttributes) as $key) {
      if ($key !== "arrayOfAttributes" && !isset($this->$key)) {
        throw new ExceptionImpots("L'attribut [$key] de la classe "
          . get_class($this) . " n'a pas été initialisé");
      }
    }
  }

  public function setFromArrayOfAttributes(array $arrayOfAttributes) {
    // يتم تهيئة بعض سمات الفئة
    foreach ($arrayOfAttributes as $key => $value) {
      $this->$key = $value;
    }
    // إرجاع الكائن
    return $this;
  }

  // toString
  public function __toString() {
    // سمات الكائن
    $arrayOfAttributes = \get_object_vars($this);
    // يتم إزالة السمة من الفئة الأم
    unset($arrayOfAttributes["arrayOfAttributes"]);
    // سلسلة Json للكائن
    return \json_encode($arrayOfAttributes, JSON_UNESCAPED_UNICODE);
  }

  // مُستخرج
  public function getArrayOfAttributes() {
    return $this->arrayOfAttributes;
  }

}

<?php

// مساحة الأسماء
namespace Application;

trait TraitDao {

  // قراءة بيانات دافعي الضرائب
  public function getTaxPayersData(string $taxPayersFilename, string $errorsFilename): array {
    // يتم استرداد بيانات المكلفين في جدول
    $baseEntity = new BaseEntity();
    $baseEntity->setFromJsonFile($taxPayersFilename);
    $arrayOfAttributes = $baseEntity->getArrayOfAttributes();
    // جدول بيانات دافعي الضرائب
    $taxPayersData = [];
    // جدول الأخطاء
    $errors = [];
    // التكرار على جدول سمات العناصر من النوع [TaxPayerData]
    $i = 0;
    foreach ($arrayOfAttributes as $attributesOfTaxPayerData) {
      // التحقق
      $error = $this->check($attributesOfTaxPayerData);
      if (!$error) {
        // مكلف ضريبي بـ +
        $taxPayersData[] = (new TaxPayerData())->setFrOmArrayOfAttributes($attributesOfTaxPayerData);
      } else {
        // خطأ في + - يتم تدوين رقم البيانات غير الصالحة
        $error = ["numéro" => $i] + $error;
        $errors[] = $error;
      }
      // التالي
      $i++;
    }
    // يتم حفظ الأخطاء في ملف json
    $string = "";
    foreach ($errors as $error) {
      $string .= \json_encode($error, JSON_UNESCAPED_UNICODE) . "\n";
    }
    $this->saveString($errorsFilename, $string);
    // نتيجة الدالة
    return $taxPayersData;
  }

  private function check(array $attributesOfTaxPayerData): array {
    // يتم التحقق من بيانات [$taxPayerData]
    // قائمة السمات التي تحتوي على أخطاء
    $attributes = [];
    // يجب أن يكون الحالة الاجتماعية نعم أو لا
    $marié = trim(strtolower($attributesOfTaxPayerData["marié"]));
    $erreur = ($marié !== "oui" and $marié !== "non");
    if ($erreur) {
      // يُلاحظ وجود خطأ
      $attributes[] = ["marié" => $marié];
    }
    // يجب أن يكون عدد الأطفال عددًا صحيحًا موجبًا أو صفرًا
    $enfants = trim($attributesOfTaxPayerData["enfants"]);
    if (!preg_match("/^\d+$/", $enfants)) {
      // تم تسجيل الخطأ
      $erreur = TRUE;
      $attributes[] = ["enfants" => $enfants];
    } else {
      $enfants = (int) $enfants;
    }

    // يجب أن يكون الراتب عددًا صحيحًا موجبًا أو صفرًا (بدون سنتات اليورو)
    $salaire = trim($attributesOfTaxPayerData["salaire"]);
    if (!preg_match("/^\d+$/", $salaire)) {
      // تم تسجيل الخطأ
      $erreur = TRUE;
      $attributes[] = ["salaire" => $salaire];
    } else {
      $salaire = (int) $salaire;
    }

    // خطأ؟
    if ($erreur) {
      // العودة مع وجود خطأ
      return ["erreurs" => $attributes];
    } else {
      // العودة بدون خطأ
      return [];
    }
  }

  public function saveResults(string $resultsFilename, array $taxPayersData): void {

    // إذا لم يكن الملف النصي [$resultsFileName] موجودًا، فسيتم إنشاؤه

    $string = "[" . implode(",
", $taxPayersData) . "]";
    // تسجيل هذه السلسلة
    $this->saveString($resultsFilename, $string);
  }

  // تسجيل نتائج جدول في ملف نصي
  private function saveString(string $fileName, string $data): void {
    // تسجيل السلسلة [$data] في الملف النصي [$fileName]
    // إذا لم يكن الملف النصي [$fileName] موجودًا، يتم إنشاؤه

    }
  }
