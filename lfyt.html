    المصطلحات: الفئة (class) هي نوع PHP. والمتغير من هذا النوع يُسمى كائنًا. والكائن هو مثيل (نسخة) من الفئة.

    تعليقات

    • السطر 6: تشير الصيغة $obj->attr إلى السمة attr للمتغير $obj. وإذا لم يكن موجودًا، يتم إنشاؤه، مما يجعل المتغير $obj كائنًا له سمات. وقد رأينا أن PHP ينشئ افتراضيًا كائنًا من النوع stdClass؛
    • السطر 16: التعبير $obj2=$obj1، عندما يكون $obj1 كائنًا، يمثل نسخًا للكائنات بالرجوع: $obj2 و $obj1 هما مرجعان (عنوانان) لكائن واحد. ويمكن تعديل الكائن نفسه بواسطة أي من هذين المرجعين؛
    • السطور 23-27: تهدف إلى إظهار أن $obj1 و $obj2 متغيران مختلفان: فهما لا يقعان في نفس عنوان الذاكرة:
      • $obj2=$obj1 نسخ قيمة $obj1 إلى المتغير $obj2 (العملية 1 أعلاه). قيمة $obj1 هي عنوان أحد الكائنات. وبالتالي، فإن $obj1 و$obj2 يشيران إلى نفس الكائن. عند التعامل مع المتغير $obj الذي يشير إلى كائن ما، فإن المتغير PHP يتعامل مع الكائن الذي يشير إليه المتغير $obj. وفقًا للرسم التخطيطي أدناه، نرى أنه يمكن تعديل الكائن المشار إليه إما عبر $obj1 أو عبر $obj2. وهذا ما تظهره السطران 4 و5 من النتائج؛

    • السطر 30: التعبير $obj3=&$obj2 يجعل $obj2 و $obj3 في نفس العنوان [1 ci-dessous]. يمكن القول إن المتغيرين هما اسمان بديلان لنفس موقع الذاكرة. كلاهما يشير إلى كائن واحد، وهو «الكائن أ» الموضح أدناه [2]؛
      • تؤدي العملية $obj2=new stdClass() إلى إنشاء كائن جديد، «الكائن ب» ([3 ci-dessous])، ويتم تعيين عنوان هذا الكائن الجديد للمتغير $obj2. ونظرًا لأن $obj2 و $obj3 هما اسمان مستعاران لنفس موقع الذاكرة، فإن $obj3 يشير أيضًا إلى الكائن الجديد «الكائن ب». وهذا ما توضحه الأسطر 16-27 و30-41 من النتائج؛

    • الأسطر 52-54: توضح أنه يمكن تصفح الكائن كقاموس. مفاتيح القاموس هي أسماء السمات وقيم القاموس، وهي قيم هذه السمات نفسها؛
    • السطر 51: يمكن تطبيق الدالة count على كائن (مع ظهور تحذير)، لكنها لا تعطي، كما كان متوقعًا، عدد السمات. لذا، فإن الكائن يشبه القاموس ولكنه ليس قاموسًا؛

    5.3. فئة «Personne» بدون سمات مُعلنة

    النص البرمجي [classes-02.php] هو التالي:

    
<?php

class Personne {

  // سمات الفئة
  // غير مُعلنة - يمكن إنشاؤها ديناميكيًا
  // طريقة
  function identite() {
    // في البداية، تستخدم سمات غير موجودة
    return "[$this->prenom,$this->nom,$this->age]";
  }

}

// اختبار
// السمات عامة ويمكن إنشاؤها ديناميكيًا
$p = new Personne();
$p->prenom = "Paul";
$p->nom = "Langevin";
$p->age = 48;
// استدعاء طريقة
print "personne=" . $p->identite() . "\n";
// نهاية
exit;

    النتائج:

    1

    تعليقات

    • الأسطر 3-13: تحدد فئة Personne. الفئة هي قالب يُستخدم لإنشاء كائنات. وهي تضم سمات ووظائف تُسمى طرقًا. ليس من الضروري إعلان السمات؛
    • الأسطر 8-11: تعرض الطريقة identité قيمة ثلاث سمات غير مُعلنة في الفئة. تشير الكلمة الرئيسية $this إلى الكائن الذي تُطبق عليه الطريقة؛
    • السطر 17: يتم إنشاء كائن $p من النوع Personne. تُستخدم الكلمة الرئيسية new لإنشاء كائن جديد. تُرجع العملية مرجعًا إلى الكائن الذي تم إنشاؤه (أي عنوانًا). يمكن كتابتها بعدة طرق: new Personne()، new Personne، new personne. اسم الفئة غير حساس لحالة الأحرف؛
    • السطور 18-20: يتم إنشاء السمات الثلاث اللازمة للطريقة identité في الكائن $p؛
    • السطر 22: يتم تطبيق الأسلوب identité الخاص بالفئة Personne على الكائن $p. في كود (الأسطر 8-11) للطريقة identité، تشير $this إلى نفس الكائن الذي تشير إليه $p؛

    5.5. فئة Personne مع منشئ

    أظهرت الأمثلة السابقة فئات Personne غير تقليدية كما يمكن العثور عليها في PHP 4. لا يُنصح باتباع هذه الأمثلة. نقدم الآن فئة Personne [classes-04.php] تتوافق مع الممارسات الجيدة في PHP 7:

    
<?php

// الالتزام الصارم بأنواع المعلمات المعلنة للدوال
declare (strict_types=1);
class Personne {
// سمات الفئة
  private $prenom;
  private $nom;
  private $age;

// دالات القراءة والكتابة
  public function getPrenom(): string {
    return $this->prenom;
  }

  public function getNom(): string {
    return $this->nom;
  }

  public function getAge(): int {
    return $this->age;
  }

  public function setPrenom(string $prenom): void {
    $this->prenom = $prenom;
  }

  public function setNom(string $nom): void {
    $this->nom = $nom;
  }

  public function setAge(int $age): void {
    $this->age = $age;
  }

// منشئ
  public function __construct(string $prenom, string $nom, int $age) {
    // نستخدم وظائف التعيين
    $this->setPrenom($prenom);
    $this->setNom($nom);
    $this->setAge($age);
  }

// طريقة toString
  public function __toString(): string {
    return "[$this->prenom,$this->nom,$this->age]";
  }

}

// اختبار
// إنشاء كائن «شخص»
$p = new Personne("Paul", "Langevin", 48);
// هوية هذا الشخص
print "personne=$p\n";
// تغيير العمر
$p->setAge(14);
// هوية الشخص
print "personne=$p\n";
// النهاية
exit;

    • public: تجعل السمة مرئية من خارج الفئة،
    • protected: يجعل السمة protégé مرئية من داخل الفئة والفئات المشتقة منها؛
    • نظرًا لأن السمات خاصة، فلا يمكن الوصول إليها من خارج الفئة. وبالتالي، لا يمكن كتابة الكود التالي:

    • تمت إضافة الكلمة الرئيسية public كبادئة لجميع طرق الفئة (الوظائف) لتشير إلى أن الوظيفة مرئية خارج الفئة. الكلمات الرئيسية الأخرى التي يمكن استخدامها هي، كما هو الحال بالنسبة للسمات وبنفس المعنى، private و protected. بدون سمة رؤية صريحة، تكون الوظيفة ذات رؤية ضمنية عامة؛

    5.6. فئة Personne مع عناصر التحقق من الصحة في المنشئ

    يُعد منشئ الفئة المكان المناسب للتحقق من صحة قيم تهيئة الكائن. لكن يمكن أيضًا تهيئة الكائن من خلال أساليبه set أو ما يعادلها. لتجنب وضع نفس عمليات التحقق في مكانين مختلفين، يمكن وضعها في الأساليب set. إذا تبين أن إحدى قيم التهيئة للكائن غير صحيحة، فسيتم إلقاء استثناء. وفيما يلي مثال على ذلك.

    نقوم أولاً بنقل تعريف الفئة Personne إلى ملفها الخاص [Personne.php]:

    
<?php

// الالتزام الصارم بأنواع المعلمات المعلنة للدوال
declare (strict_types=1);

// مساحة الأسماء؛
namespace Exemples;

// فئة «Personne»
class Personne {
// سمات الفئة
  private $prenom;
  private $nom;
  private $age;

// دالات الاسترجاع والتعيين
  public function getPrenom(): string {
    return $this->prenom;
  }

  public function getNom(): string {
    return $this->nom;
  }

  public function getAge(): int {
    return $this->age;
  }

  public function setPrenom(string $prénom): void {
    // يجب ألا يكون الاسم الأول فارغًا
    $prénom = trim($prénom);
    if ($prénom === "") {
      throw new \Exception("Le prénom doit être non vide");
    } else {
      $this->prenom = $prénom;
    }
  }

  public function setNom(string $nom): void {
    // يجب ألا يكون الاسم فارغًا
    $nom = trim($nom);
    if ($nom === "") {
      throw new \Exception("Le nom doit être non vide");
    } else {
      $this->nom = $nom;
    }
  }

  public function setAge(int $âge): void {
    // يجب أن يكون العمر صالحًا
    if ($âge < 0) {
      throw new \Exception("L'âge doit être un entier positif ou nul");
    } else {
      $this->age = $âge;
    }
  }

// منشئ
  public function __construct(string $prenom, string $nom, int $age) {
    // نستخدم طرق التعيين
    $this->setPrenom($prenom);
    $this->setNom($nom);
    $this->setAge($age);
  }

  // طريقة
  public function initWithPersonne(Personne $p): void {
    // تُهيئ الكائن الحالي بشخص ما $p
    $this->__construct($p->prenom, $p->nom, $p->age);
  }

  // طريقة toString
  function __toString(): string {
    return "[$this->prenom,$this->nom,$this->age]";
  }

}

    5.14. الفئات المجردة

    الفئة المجردة هي فئة غير مكتملة لا يمكن إنشاء مثيل لها. ويجب بالضرورة أن تكون مشتقة حتى يمكن استخدامها.

    ما فائدة الفئة المجردة؟ أحيانًا تكون لدينا فئات تشترك في طريقة أو أكثر ولكنها تختلف في طرق أو سمات أخرى. ومن المستحسن في هذه الحالة تجميع كل ما هو مشترك في فئة أصلية. في الوقت الحالي، لا نحتاج إلى فئة مجردة. لكن لنفترض أن الفئات الفرعية لا تختلف إلا في طريقة M: سيكون توقيع الطريقة هو نفسه في جميع الفئات الفرعية، لكن تنفيذها سيختلف. لإجبار الفئات الفرعية على تنفيذ الطريقة M:

    • سنقوم بإعلان توقيع الطريقة M في الفئة الأم. وبما أن هذه الفئة لا تعرف كيفية تنفيذها، فإننا نضع الكلمة المفتاحية abstract كبادئة للطريقة: وهذا يعني أن تنفيذ الطريقة M يُحال إلى الفئات الفرعية؛
    • ونظرًا لأن الفئة الأم ليست مُنفَّذة بالكامل، فإنها تُعلن أيضًا على أنها مجردة باستخدام نفس الكلمة المفتاحية abstract. وهذا يجعل من المستحيل إنشاء مثيل للفئة. لذا، من الضروري إنشاء فئة فرعية تحدد تنفيذ الدالة M، حتى يصبح جسم الفئة الأم قابلاً للاستخدام؛

<?php

  public function f(): int {
    return 1;

  abstract function g(): int;

  }

    5.19. ترميز فئة بـ jSON

    غالبًا ما توجد الطريقة [__toString] في الفئة: ومن المفترض أن تُرجع سلسلة أحرف تمثل الكائن الذي يستدعيها. قد يكون من المغري أن تكون هذه السلسلة سلسلة jSON. سنستكشف هذا المسار الآن.

class Personne {
  // السمات
  private $nom;
  private $prénom;
  private $âge;
  private $enfants;

  // مُعيّن عام
  public function setFromArray(array $arrayOfAttributes): Personne {

      $this->$attribute = $value;
    }
    // إرجاع الكائن
    return $this;
  }

  public function getNom() {
    return $this->nom;
  }

  public function getPrénom() {
    return $this->prénom;
  }

  public function getÂge() {
    return $this->âge;
  }

  public function getEnfants() {
    return $this->enfants;
  }

  // __toString
  public function __toString(): string {
    // تحديد الكائن
    var_dump($this);
    // استرداد سماته
    $attributes = \get_object_vars($this);
    var_dump($attributes);

  // __toString
  public function __toString(): string {
    // يُرجع سلسلة JSON من السمات
    $attributes = [];
    $this->getRecursiveAttributes($attributes, $this, []);
    // السلسلة JSON من السمات

  public function getAttributes(): array {
    return \get_object_vars($this);
  }

  private function getRecursiveAttributes(array &$attributes, $value, $keys): void {
    // تحليل القيمة [$value]
    // $keys عبارة عن مصفوفة [key1, key2, .., keyn]
    // $value=$attributesQZXW2HTMLBW2tleTFdZQXQZXW2HTMLBW2tleTJdZQX….[keyn]
    // إذا كان [$value] كائنًا، فإننا نستخدم طريقة [getAttributes] الخاصة به
    if (\is_object($value)) {
      // سمات الكائن [$value]
      $objectAttributes = $value->getAttributes();
      // ماذا نفعل بالنتيجة؟
      if ($keys) {
        // في [$attributes]، سنستبدل $value بالجدول الخاص بسماته
        // يجب إنشاء العنصر $attributesQZXW2HTMLBW2tleTFdZQXQZXW2HTMLBW2tleTJdZQX…[keyn]
        // حيث $keys هو المصفوفة [key1, key2, .., keyn]
        // يتم أخذ مرجع المصفوفة [$attributes]
        $attribute = &$attributes;
        // نقوم بمسح جدول المفاتيح
        foreach ($keys as $key) {
          // نأخذ مرجع السمة
          $attribute = &$attribute[$key];
        }
        // هنا $attribut و $attributesQZXW2HTMLBW2tleTFdZQXQZXW2HTMLBW2tleTJdZQX…[key(n)] متطابقة
        // وهي تشترك في نفس موقع الذاكرة
        // يتم استبدال الكائن [$value] بمصفوفة سماته؛
        // يجب كتابة $attributesQZXW2HTMLBW2tleTFdZQXQZXW2HTMLBW2tleTJdZQX…[keyn]=$objectAttributes
        // وهو ما يعادل $attribute = $objectAttributes
        $attribute = $objectAttributes;
      } else {
        // لا توجد مفاتيح - نحن في بداية استكشاف الكائن
        // $objectAttributes يمثل سمات المستوى الأول للفئة
        $attributes += $objectAttributes;
      }
      // ربما لا تزال هناك كائنات في [$objectAttributes]
      // نستكشف سمات [$objectAttributes]
      $this->getRecursiveAttributes($attributes, $objectAttributes, $keys);
    } else {
      if (\is_array($value)) {
        // لدينا مصفوفة - نقوم بتحليل كل عنصر من عناصرها
        foreach ($value as $key => $élément) {
          // نضيف المفتاح الحالي إلى المصفوفة $keys
          \array_push($keys, $key);
          // نقوم بتحليل $élément
          $this->getRecursiveAttributes($attributes, $élément, $keys);
          // نحذف المفتاح الذي تم تحليله للتو من المصفوفة $keys
          \array_pop($keys);
        }
      }
    }
  }
