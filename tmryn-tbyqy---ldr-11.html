    22. تمرين تطبيقي – الإصدار 11

    لا يزال من الشائع أن ترسل خدمات الويب استجاباتها في شكل تدفق XML بدلاً من تدفق jSON:

    • تدفق jSON أخف وزنًا، لكنه يتطلب دليلًا لاستخدامه لفهمه؛
    • أما تدفق XML فهو أكثر تفصيلاً ولكنه موثق ذاتيًا، ويمكن فهمه على الفور؛

    نقوم بتعديل الإصدار 11 من العميل/الخادم بحيث يرسل الخادم الآن تدفقًا XML كرد على عملائه:

  // من https://stackoverflow.com/questions/1397036/how-to-convert-array-to-simplexml
  public static function getXmlForArrayOfAttributes(array $arrayOfAttributes,
    \SimpleXmlElement &$node): void {
    // يتم مسح سمات المصفوفة
    foreach ($arrayOfAttributes as $attribute => $value) {
      // هل السمة رقمية؟
      if (is_numeric($attribute)) {
        // حالة مؤشر المصفوفة (وكذلك حالات أخرى)
        $attribute = 'i' . $attribute;
      }
      // هل $value مصفوفة؟
      if (is_array($value)) {
        // سنستكشف المصفوفة [$value] بدورها
        // نضيف عقدة إلى الرسم البياني XML
        $subnode = $node->addChild($attribute);
        // استدعاء متكرر لاستكشاف المصفوفة [$value]
        Utilitaires::getXmlForArrayOfAttributes($value, $subnode);
      } else {
        // يتم إضافة العقدة إلى الرسم البياني XML
        $node->addChild("$attribute", htmlspecialchars("$value"));
      }
    }
  }
}

// وظيفة إرسال الإجابة HTTP إلى العميل
function sendResponse(Response $response, array $result, int $statusCode,
  array $headers, Logger $logger = NULL, \Predis\Client $predisClient = NULL) {
  // $response: الإجابة HTTP
  // $result: جدول النتائج
  // $statusCode: حالة HTTP للرد
  // $headers: رؤوس HTTP المراد إدراجها في الرد
  // $logger: مسجل سجلات التطبيق
  // $predisClient: عميل [predis]
  //
  // الحالة HTTTP
  $response->setStatusCode($statusCode);
  // النص الأساسي XML
  $node = new \SimpleXMLElement("<?xml version='1.0' encoding='UTF-8'?><réponse></réponse>");
  Utilitaires::getXmlForArrayOfAttributes($result, $node);
  $response->setContent($node->asXML());
  // رؤوس الرسالة
  $response->headers->add($headers);
  // الإرسال
  $response->send();
  // السجل
  if ($logger != NULL) {
    // السجل في jSON
    $log = \json_encode(["réponse" => $result], JSON_UNESCAPED_UNICODE);
    $logger->write("$log\n");
    $logger->close();
  }
  // إغلاق الاتصال [redis]
  if ($predisClient != NULL) {
    $predisClient->disconnect();
  }
}

    22.2.1. الطبقة [dao]

    يتم تعديل العميل [ClientDao.php] (السطر 13 أعلاه) لمراعاة التنسيق الجديد للاستجابة. يتم استخدام [simpleXML] لمعالجة هذه الاستجابة:

    
<?php

namespace Application;

// التبعيات
use \Symfony\Component\HttpClient\HttpClient;

class ClientDao implements InterfaceClientDao {
  // استخدام سمة
  use TraitDao;

  private $urlServer;
  private $user;
  private $sessionCookie;

    $this->urlServer = $urlServer;
    $this->user = $user;
  }

  // حساب الضريبة
  public function calculerImpot(string $marié, int $enfants, int $salaire): array {
    …
    // استرداد الرد XML
    $réponse = $response->getContent(false);
    $xml = new \SimpleXMLElement($réponse);
    // السجلات
    // طباعة "$réponse\n";
    // نحصل على حالة الرد
    $statusCode = $response->getStatusCode();
    // هل هناك خطأ؟
    if ($statusCode !== 200) {
      // هناك خطأ - يتم إلقاء استثناء
      $message = \json_encode(["statut HTTP" => $statusCode, "réponse" => $xml], JSON_UNESCAPED_UNICODE);
      throw new ExceptionImpots($message);
    }
    if (!$this->sessionCookie) {
      // استرداد ملف تعريف الارتباط الخاص بالجلسة
      $headers = $response->getHeaders();
      if (isset($headers["set-cookie"])) {
        // ملف تعريف ارتباط الجلسة؟
        foreach ($headers["set-cookie"] as $cookie) {
          $match = [];
          $match = preg_match("/^PHPSESSID=(.+?);/", $cookie, $champs);
          if ($match) {
            $this->sessionCookie = "PHPSESSID=" . $champs[1];
          }
        }
      }
    }
    // نقوم بإرجاع الرد في شكل مصفوفة
    return \json_decode(\json_encode($xml, JSON_UNESCAPED_UNICODE), true);
  }

}
